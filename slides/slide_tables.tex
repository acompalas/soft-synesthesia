% ---------------------------------------------------------------- 1. reward
\begin{slide}
\textbf{Reward at every control step} (10 Hz, episode up to 500 steps = 50 s)
\[
r_t \;=\; \underbrace{\Delta\theta_z}_{\text{turn about vertical}}
\;-\; \underbrace{0.1\,\lVert \mathbf v_{\text{obj}} \rVert}_{\text{don't slide it}}
\;+\; \underbrace{0.1\, r_{\text{near}}}_{\text{fingertips close}}
\;-\; \underbrace{3{\times}10^{-4}\textstyle\sum_j \tau_j^2}_{\text{torque}}
\;-\; \underbrace{3{\times}10^{-4}\textstyle\sum_j |\tau_j\,\dot q_j|}_{\text{work}}
\;-\; \underbrace{50\cdot\mathbf{1}[\text{fall}]}_{\text{dropped}}
\]
\color{muted}\small
$\Delta\theta_z$ is signed: turning the wrong way costs reward.
Episode ends (``fall'') if the object moves $>10$\,cm from its start in the hand, tilts $>72^\circ$, or slides off.\\
Cumulative reward $=\sum_t r_t$ over an episode; time to fall (TTF) = seconds until a fall, at most 50\,s.
\end{slide}
